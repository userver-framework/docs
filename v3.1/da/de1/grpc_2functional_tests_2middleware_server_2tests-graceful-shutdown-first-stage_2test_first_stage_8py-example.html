137def check_present(metadata, headers: dict[str, list[str]]):
138 metadata_dict = to_dict(metadata)
139 for k, v in headers.items():
140 assert k in metadata_dict
141 assert metadata_dict[k] == v

144def check_not_present(metadata, headers: dict[str, list[str]]):
145 metadata_dict = to_dict(metadata)
146 for k in headers.keys():
147 assert k not in metadata_dict

150def to_dict(metadata) -> dict[str, list[str]]:
151 result: dict[str, list[str]] = {}
152 for k, v in metadata:
153 result.setdefault(k, []).append(v)
154
155 return result
