188TEST_P(AnyToJsonSuccessTest, Test) {
189 const auto& param = GetParam();
190
191 auto input = PrepareTestData(param.input);
192 formats::json::Value json;
193 formats::json::Value expected_json;
194 formats::json::Value sample_json;
195
196 UASSERT_NO_THROW((json = MessageToJson(input, param.options)));
197 UASSERT_NO_THROW((expected_json = PrepareJsonTestData(param.expected_json)));
198
199 if (!param.skip_native_check) {
200 UASSERT_NO_THROW((sample_json = CreateSampleJson(input, param.options)));
201 EXPECT_EQ(json, sample_json);
202 }
203
204 EXPECT_EQ(json, expected_json);
205}

246TEST(AnyToJsonAdditionalTest, DynamicMessage) {
247 using Message = ::google::protobuf::Any;
248 using ProtobufStringType =
249 decltype(std::declval<::google::protobuf::Reflection>()
250 .GetString(std::declval<const ::google::protobuf::Message&>(), nullptr));
251
252 ::google::protobuf::DynamicMessageFactory factory;
253 proto_json::messages::Int32Message payload;
254 ProtobufStringType payload_str;
255
256 payload.set_field1(1001);
257 payload.set_field2(1002);
258 payload.set_field3(1003);
259
260 ASSERT_TRUE(payload.SerializeToString(&payload_str));
261 proto_json::messages::Int32Message payload2;
262
263 {
264 std::unique_ptr<::google::protobuf::Message> message(factory.GetPrototype(Message::descriptor())->New());
265 const auto reflection = message->GetReflection();
266 const auto type_url_desc = message->GetDescriptor()->FindFieldByName("type_url");
267 const auto value_desc = message->GetDescriptor()->FindFieldByName("value");
268
269 reflection->SetString(message.get(), type_url_desc, "type.googleapis.com/proto_json.messages.Int32Message");
270 reflection->SetString(message.get(), value_desc, payload_str);
271
272 formats::json::Value json;
273
274 UASSERT_NO_THROW((json = MessageToJson(*message, {})));
275 ASSERT_TRUE(json.IsObject());
276 ASSERT_EQ(json.GetSize(), std::size_t{4});
277 EXPECT_EQ(json["@type"].As<std::string>(), "type.googleapis.com/proto_json.messages.Int32Message");
278 EXPECT_EQ(json["field1"].As<std::int32_t>(), 1001);
279 EXPECT_EQ(json["field2"].As<std::int32_t>(), 1002);
280 EXPECT_EQ(json["field3"].As<std::int32_t>(), 1003);
281 }
