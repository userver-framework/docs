21
22 if (!request.body.empty()) {
23 header_content_md5 = MakeHeaderContentMd5(request.body);

26 auto header_date = MakeHeaderDate();

28
29 std::unordered_map<std::string, std::string> auth_headers{
30 {kDate, std::move(header_date)},
