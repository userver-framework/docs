REDIS_DEFAULT_COMMAND_CONTROL Dynamic Config

Dynamic config that overrides the default timeouts, number of retries and server selection strategy for redis commands.

Schema:

# yaml
default: {}
description: |
Dynamic config that overrides the default timeouts, number of retries and
server selection strategy for redis commands.
schema:
example: |
{
"best_dc_count": 0,
"max_ping_latency_ms": 0,
"max_retries": 4,
"strategy": "default",
"timeout_all_ms": 2000,
"timeout_single_ms": 500
}
additionalProperties: false
properties:
best_dc_count:
type: integer
max_ping_latency_ms:
type: integer
max_retries:
type: integer
strategy:
enum:
- default
- every_dc
- local_dc_conductor
- nearest_server_ping
type: string
timeout_all_ms:
type: integer
timeout_single_ms:
type: integer
force_request_to_master:
type: boolean
allow_reads_from_master:
type: boolean
consider_ping:
type: boolean
type: object
