12/// @brief Class that can be escaped for clickhouse queries instead of [double] and [float].
13/// [double] and [float] are escaped as FloatingWithPrecision<double, 12> and
14/// FloatingWithPrecision<float, 6> respectively. These constants were derived from the accuracity the numbers from
15/// ranges [0.0-100000.0] (doubles) and [0.0-100.0] (floats) can be represented by these type. Clickhouse also supports
16/// inf, -inf and nan, see clickhouse docs.
17template <
18 typename FloatingPointT,
19 std::uint32_t Precision = std::is_same_v<FloatingPointT, float> ? 6 : 12,
20 typename = std::enable_if_t<std::is_floating_point_v<FloatingPointT>>>
21class FloatingWithPrecision {
22public:
23 template <typename U>
24 FloatingWithPrecision(U value)
25 : value_(value)
26 {}
27
28 template <typename U, std::uint32_t AnotherPrecision>
29 FloatingWithPrecision(const FloatingWithPrecision<U, AnotherPrecision>& other)
30 : value_(other.value_)
31 {}
32
33 template <typename U, std::uint32_t AnotherPrecision>
34 FloatingWithPrecision(FloatingWithPrecision<U, AnotherPrecision>&& other)
35 : value_(other.value_)
36 {}
37
38 template <typename U, std::uint32_t AnotherPrecision>
39 FloatingWithPrecision& operator=(const FloatingWithPrecision<U, AnotherPrecision>& other) {
40 value_ = other.value_;
41 return *this;
42 }
43
44 template <typename U, std::uint32_t AnotherPrecision>
